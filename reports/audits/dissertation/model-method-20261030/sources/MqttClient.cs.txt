0256         public virtual bool Publish(string topic, byte[] payload, MqttQos qos, bool retain = false)
0257         {
0258             ArgumentNullException.ThrowIfNull(payload);
0259             var localPublishSequence = Interlocked.Increment(ref _localPublishSequence);
0260             ushort? packetId = qos == MqttQos.AtMostOnce ? null : (ushort?)_packetIdProvider.GetPacketId();

0262             {
0263                 Payload = payload
0264             };
0265             var startedTick = NetworkSimulator!.TotalTicks;
0266             if (PublicationOutcome != null && packetId.HasValue)
0267                 (observedPublications ??= new())[packetId.Value] = (payload, startedTick);

0270                 NodeType.Server,
0271                 ServerIndex ?? -1,
0272                 _packetManager.PacketToBytes(publish), ProtocolType,
0273                 new PublishIdentity(localPublishSequence, NetworkSimulator.TotalTicks), "MQTT Publish");
0274             var scope = NetworkSimulator!.Monitoring.WithBeginScope(NetworkSimulator.TotalTicks, ref packetPayload!,
0275                 $"Publish {Name} to {packetPayload.To} with topic {topic}");
0276 
0277             if (!IsConnected)
0278             {
0279                 SaveMessage(publish, Name, PendingMessageStatus.PendingToSend);
0280                 return true;
0281             }
0282 
0283             if (qos > MqttQos.AtMostOnce)
0284             {
0285                 var status = qos == MqttQos.ExactlyOnce
0286                     ? PendingMessageStatus.AwaitingPublishReceived
0287                     : PendingMessageStatus.PendingToAcknowledge;
0288                 SaveMessage(publish, Name, status);
0289             }
0290             _mqttCounters.Increment(publish.Type);
0291 
0292             var sendResult = Send(packetPayload);
0293 
0294             if (!sendResult)
0295             {
0296                 _mqttCounters.Refuse(publish.Type);
0297                 if (packetId.HasValue) ObservePublicationOutcome(packetId.Value, false, "outgoingSendRefused");
0298             }
0299             if (qos == MqttQos.AtMostOnce)
0300                 PublicationOutcome?.Invoke(this, new(payload, startedTick, NetworkSimulator.TotalTicks, sendResult,
0301                     sendResult ? null : "outgoingSendRefused"));
0302 
0303             return sendResult;
0304         }

0321         protected override void OnReceived(INetworkPacket packet)
0322         {
0323             var payload = _packetManager.BytesToPacket<PacketBase>(packet.Payload);
0324             if (payload == null)
0325             {
0326                 base.OnReceived(packet);
0327             }
0328 
0329             NetworkSimulator!.Monitoring.WithEndScope(NetworkSimulator.TotalTicks, ref packet);
0330             switch (payload!.Type)
0331             {
0332                 case MqttPacketType.PublishAck:
0333                     ProcessPublishAckFromBroker(packet, _packetManager.BytesToPacket<PublishAckPacket>(packet.Payload));
0334                     return;
0335                 case MqttPacketType.PublishReceived:
0336                     ProcessPublishReceivedFromBroker(packet,
0337                         _packetManager.BytesToPacket<PublishReceivedPacket>(packet.Payload));
0338                     return;
0339                 case MqttPacketType.PublishRelease:
0340                     ProcessPublishReleaseFromBroker(packet,
0341                         _packetManager.BytesToPacket<PublishReleasePacket>(packet.Payload));
0342                     return;
0343                 case MqttPacketType.PublishComplete:
0344                     ProcessPublishCompleteFromBroker(packet,
0345                         _packetManager.BytesToPacket<PublishCompletePacket>(packet.Payload));
0346                     return;
0347                 case MqttPacketType.PingResponse:
0348                     break;
0349                 case MqttPacketType.Publish:
0350                     ProcessPublishFromBroker(packet, _packetManager.BytesToPacket<PublishPacket>(packet.Payload));
0351                     return;
0352                 case MqttPacketType.ConnectAck:
0353                     ProcessConnectAckFromBroker(packet, _packetManager.BytesToPacket<ConnectAckPacket>(packet.Payload));
0354                     return;
0355                 case MqttPacketType.SubscribeAck:
0356                     ProcessSubscribeAckFromBroker(packet, _packetManager.BytesToPacket<SubscribeAckPacket>(packet.Payload));
0357                     return;
0358                 default:
0359                     return;
0360             }
0361 
0362             _mqttCounters.PacketTypeCounters[payload.Type].Increment();
0363         }

0378         private void ProcessPublishFromBroker(INetworkPacket packet, PublishPacket? publishPacket)
0379         {
0380             if (publishPacket == null)
0381             {
0382                 return;
0383             }
0384 
0385             if (publishPacket.QualityOfService == MqttQos.ExactlyOnce)
0386             {

0389                     message.PacketId == publishPacket.PacketId &&
