0110         protected override void OnReceived(INetworkPacket packet)
0111         {
0112             var payload = _packetManager.BytesToPacket<PacketBase>(packet.Payload);
0113             if (payload == null)
0114             {
0115                 base.OnReceived(packet);
0116             }
0117             NetworkSimulator!.Monitoring.WithEndScope(NetworkSimulator.TotalTicks, ref packet);
0118             switch (payload!.Type)
0119             {
0120                 case MqttPacketType.Publish:
0121                     ProcessFromClientPublish(packet, _packetManager.BytesToPacket<PublishPacket>(packet.Payload));
0122                     break;
0123                 case MqttPacketType.PublishReceived:
0124                     ProcessFromClientPublishReceived(packet,
0125                         _packetManager.BytesToPacket<PublishReceivedPacket>(packet.Payload));
0126                     break;
0127                 case MqttPacketType.PublishRelease:
0128                     ProcessFromClientPublishRelease(packet, _packetManager.BytesToPacket<PublishReleasePacket>(packet.Payload));
0129                     break;
0130                 case MqttPacketType.PublishComplete:
0131                     ProcessFromClientPublishComplete(packet,
0132                         _packetManager.BytesToPacket<PublishCompletePacket>(packet.Payload));
0133                     break;
0134                 case MqttPacketType.PingRequest:
0135                     break;
0136                 case MqttPacketType.PublishAck:
0137                     ProcessToClientPublishAck(packet, _packetManager.BytesToPacket<PublishAckPacket>(packet.Payload));
0138                     break;
0139                 case MqttPacketType.Connect:
0140                     var contextPacket = (NetworkPacket<(string Server, bool CleanSession)>)packet;
0141                     ProcessConnect(packet, _packetManager.BytesToPacket<ConnectPacket>(packet.Payload), contextPacket.TypedContext);
0142                     break;
0143                 case MqttPacketType.Disconnect:
0144                     break;
0145                 case MqttPacketType.Subscribe:
0146                     var subscribeContextPacket = (NetworkPacket<(string TopicFilter, MqttQos Qos)>)packet;
0147                     ProcessSubscribe(packet, _packetManager.BytesToPacket<SubscribePacket>(packet.Payload), subscribeContextPacket.TypedContext);
0148                     break;
0149                 case MqttPacketType.Unsubscribe:
0150                     break;
0151                 default:
0152                     return;
0153             }
0154 
0155             MqttCounters.PacketTypeCounters[payload.Type].Increment();
0156         }

0158         private void ProcessFromClientPublish(INetworkPacket packet, PublishPacket? publishPacket)
0159         {
0160             if (publishPacket == null)
0161             {
0162                 return;
0163             }
0164 
0165             var identity = packet.Context as PublishIdentity;
0166             var localPublishSequence = identity?.Sequence ?? (packet.Context is long sequence ? sequence : packet.Id);
0167             var tracker = _measurement.BeginAttempt(packet.From, publishPacket.QualityOfService,
0168                 publishPacket.PacketId, packet.Id, identity?.StartedTick ?? NetworkSimulator!.TotalTicks,
0169                 localPublishSequence, publishPacket.Payload.Length);
0170             if (tracker.Admitted)
0171             {
0172                 ProcessFromClientPublishCore(packet, publishPacket);
0173                 return;
0174             }
0175 
0176             // Обратная совместимость: без профиля брокер обрабатывает сообщения
0177             // без ограничений по RPS, задержке и отказам.
0178             if (_profile == null)
0179             {
0180                 _measurement.Admit(tracker);
0181                 ProcessFromClientPublishCore(packet, publishPacket);
0182                 return;
0183             }
0184 
0185             var clients = _profileClientCount ?? Math.Max(1, GetServerClients().Count());
0186             var sample = _profile.For(publishPacket.Payload.Length,
0187                 publishPacket.QualityOfService, clients);
0188 
0189             // 1. Лимитер RPS (пропускная способность).
0190             var limiterKey = (publishPacket.Payload.Length, publishPacket.QualityOfService, clients);
0191             var limiter = _rateLimiters.GetOrAdd(limiterKey,
0192                 _ => new BrokerRateLimiter(sample.CapacityRps, _tickPeriod));
0193             var capacityWaitTicks = EmpiricalCapacityPacing ? limiter.Reserve(NetworkSimulator!.TotalTicks) : 0;
0194             if (!EmpiricalCapacityPacing && !limiter.TryAcquire(NetworkSimulator!.TotalTicks))
0195             {
0196                 MqttCounters.RefuseByRateLimit(publishPacket.QualityOfService);
0197                 _measurement.RejectByRate(tracker);
0198                 FailClientPublish(packet.From, publishPacket.PacketId);
0199                 return;
0200             }
0201 
0202             // 2. Вероятностный отказ.
0203             if (DeterministicMqttSampler.Uniform(_randomSeed, Name, packet.From,
0204                     localPublishSequence, "publish-failure") < sample.PublishFailureRate)
0205             {
0206                 MqttCounters.RefuseByFailRate(publishPacket.QualityOfService);
0207                 _measurement.FailPublish(tracker);
0208                 FailClientPublish(packet.From, publishPacket.PacketId);
0209                 return;
0210             }
0211 
0212             // 3. Постановка в очередь обработки с задержкой.
0213             tracker.ConditionalDeliveryLossRate = sample.ConditionalDeliveryLossRate;
0214             _measurement.Admit(tracker);
0215             var uniform = DeterministicMqttSampler.Uniform(_randomSeed, Name, packet.From,
0216                 localPublishSequence, "processing-latency");
0217             var processingTicks = sample.ObservedLatencyDistribution != null && EmpiricalProtocolFloorTicks.HasValue
0218                 ? checked((int)Math.Max(0, EmpiricalTicksFromMs(sample.ObservedLatencyDistribution.Sample(uniform)) - EmpiricalProtocolFloorTicks.Value))
0219                 : sample.ProcessingLatencyDistribution != null
0220                 ? EmpiricalTicksFromMs(sample.ProcessingLatencyDistribution.Sample(uniform))
0221                 : sample.ProcessingLatencyQuantiles == null
0222                 ? 1
0223                 : TicksFromMs(DeterministicMqttSampler.InverseCdf(
0224                     uniform,
0225                     sample.ProcessingLatencyQuantiles));
0226             // The observed end-to-end CDF already includes native queueing. Capacity
0227             // eligibility is an additional lower bound, not an independent delay to add twice.
0228             _processingQueue.Enqueue(packet, Math.Max(processingTicks, capacityWaitTicks));
0229         }

0238         private void ProcessFromClientPublishCore(INetworkPacket packet, PublishPacket? publishPacket)
0239         {
0240             if (publishPacket == null)
0241             {
0242                 return;
0243             }
0244 
0245             NetworkSimulator!.Monitoring.WithEndScope(NetworkSimulator.TotalTicks, ref packet);
0246 
0247             var clientId = packet.From;
0248 
0249             ValidatePublish(clientId, publishPacket);
0250 
0251             var qos = publishPacket.QualityOfService > _maximumQualityOfService
0252                 ? _maximumQualityOfService
0253                 : publishPacket.QualityOfService;
0254 
0255             var session = _sessionRepository.Read(clientId);
0256 
0257             if (session == null)
0258             {
0259                 throw new MqttException($"Client Session {clientId} Not Found");
0260             }
0261 
0262             if (qos == MqttQos.ExactlyOnce)
0263             {
0264                 var packetId = publishPacket.PacketId!.Value;
0265                 if (session.GetPendingMessages().Any(message =>
0266                         message.Status == PendingMessageStatus.AwaitingPublishRelease &&
0267                         message.PacketId == packetId))
0268                 {
0269                     SendPublishReceived(packet, publishPacket);
0270                     return;
0271                 }
0272 
0273                 var completed = session.GetPendingAcknowledgements().FirstOrDefault(ack =>
0274                     ack.Type == MqttPacketType.PublishComplete && ack.PacketId == packetId);
0275                 session.RemovePendingAcknowledgement(completed);
0276 
0277                 SaveMessage(publishPacket, clientId, PendingMessageStatus.AwaitingPublishRelease);
0278                 SendPublishReceived(packet, publishPacket);
0279                 return;
0280             }
0281 
0282             if (qos == MqttQos.AtLeastOnce)
0283             {
0284                 SendPublishAck(packet, clientId, qos, publishPacket);
0285             }
0286 
0287             FanOutAndCount(packet, publishPacket);
0288         }

0290         private void FanOutAndCount(INetworkPacket packet, PublishPacket publishPacket)
0291         {
0292             var tracker = _measurement.Find(packet.From, publishPacket.QualityOfService,
0293                 publishPacket.PacketId, packet.Id);
0294             var subscriptions = _sessionRepository
0295                 .ReadAll()
0296                 .SelectMany(s => s.GetSubscriptions())
0297                 .Where(x => _topicEvaluator.Matches(publishPacket.Topic, x.TopicFilter))
0298                 .OrderBy(x => x.ClientId, StringComparer.Ordinal)
0299                 .ThenBy(x => x.TopicFilter, StringComparer.Ordinal)
0300                 .ToArray();
0301 
0302             foreach (var subscription in subscriptions)
0303             {
0304                 ProcessToClientPublish(packet, subscription, publishPacket, tracker);
0305             }
0306 
0307             MqttCounters.IncrementPublish(publishPacket.QualityOfService);
0308             _measurement.CompleteFanOut(tracker);
0309         }

0311         private bool ProcessToClientPublish(
0312             INetworkPacket packet, ClientSubscription subscription, PublishPacket? publishPacket,
0313             BrokerMeasurement.PublishTracker? tracker)
0314         {
0315             if (publishPacket == null)
0316             {
0317                 return false;
0318             }
0319 
0320             var supportedQos = publishPacket.QualityOfService > subscription.MaximumQualityOfService
0321                 ? subscription.MaximumQualityOfService
0322                 : publishPacket.QualityOfService;
0323 
0324             ushort? packetId = supportedQos == MqttQos.AtMostOnce ? null : (ushort?)_packetIdProvider.GetPacketId();
0325             var retain = publishPacket.Retain;
0326             var subscriptionPublish = new PublishPacket(publishPacket.Topic, supportedQos, retain, duplicated: false, packetId: packetId)
0327             {
0328                 Payload = publishPacket.Payload
0329             };
0330             var packetPayload = GetPacket(
0331                 NetworkSimulator!.GetPacketId(), subscription.ClientId, NodeType.Client, packet.FromIndex,
0332                     _packetManager.PacketToBytes(subscriptionPublish), ProtocolType, "MQTT Publish");
0333             _measurement.ExpectDelivery(tracker, packetPayload.Id);
0334 
0335             if (tracker != null && DeterministicMqttSampler.Uniform(
0336                     _randomSeed, Name, tracker.Publisher, tracker.LocalPublishSequence,
0337                     $"delivery-loss:{subscription.ClientId}") < tracker.ConditionalDeliveryLossRate)
0338             {
0339                 _measurement.DeliveryDropped(packetPayload.Id);
0340                 return true;
0341             }
0342 
0343             if (subscriptionPublish.QualityOfService > MqttQos.AtMostOnce)
0344             {
0345                 var status = subscriptionPublish.QualityOfService == MqttQos.ExactlyOnce
0346                     ? PendingMessageStatus.AwaitingPublishReceived
0347                     : PendingMessageStatus.PendingToAcknowledge;
0348                 SaveMessage(subscriptionPublish, subscription.ClientId, status);
0349             }
0350 
0351             var scope = NetworkSimulator!.Monitoring.WithBeginScope(NetworkSimulator.TotalTicks, ref packetPayload, $"Publish {Name} to  Subscriber {packetPayload.To} with topic {publishPacket.Topic}");
0352             MqttCounters.PacketTypeCounters[subscriptionPublish.Type].Increment();
0353             
0354             var sendResult = Send(packetPayload);
0355             if (!sendResult)
0356                 _measurement.DeliveryDropped(packetPayload.Id);
0357             return sendResult;
0358         }
